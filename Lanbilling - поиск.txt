// ==UserScript==
// @name         LANBilling поиск
// @namespace    uplink.extjs.combo.search
// @version      1.0.0
// @description  Добавляет поиск в открывающиеся ExtJS x-combo-list dropdown.
// @match        *://billing.uplink.kz/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==

(() => {
  'use strict';

  if (window.top !== window.self) return;

  const SEARCH_CLASS = 'uplink-extjs-combo-search';
  const MENU_SELECTOR = '.x-layer.x-combo-list';
  const INNER_SELECTOR = '.x-combo-list-inner';
  const ITEM_SELECTOR = '.x-combo-list-item';

  const wired = new WeakSet();
  let scanTimer = null;

  function visible(el) {
    if (!el || !el.isConnected) return false;

    const rect = el.getBoundingClientRect();
    if (!rect.width || !rect.height) return false;

    const style = getComputedStyle(el);
    return (
      style.display !== 'none' &&
      style.visibility !== 'hidden' &&
      Number(style.opacity) !== 0
    );
  }

  function itemLabel(item) {
    // В ExtJS видимый текст часто обрезан через "...",
    // а полное название лежит в ext:qtip.
    return (
      item.getAttribute('ext:qtip') ||
      item.getAttribute('data-qtip') ||
      item.textContent ||
      ''
    )
      .replace(/\s+/g, ' ')
      .trim();
  }

  function normalize(text) {
    return String(text || '')
      .toLocaleLowerCase()
      .replace(/\s+/g, ' ')
      .trim();
  }

  function getItems(menu) {
    return [...menu.querySelectorAll(ITEM_SELECTOR)];
  }

  function applySearch(menu, input) {
    const query = normalize(input.value);
    const items = getItems(menu);

    let shown = 0;

    for (const item of items) {
      const match = normalize(itemLabel(item)).includes(query);
      item.style.display = match ? '' : 'none';

      if (match) shown += 1;
    }

    const counter = menu.querySelector(`.${SEARCH_CLASS} .uplink-extjs-search-count`);
    if (counter) {
      counter.textContent = `${shown} из ${items.length}`;
    }

    // После фильтрации возвращаем внутренний scroll наверх,
    // чтобы первый результат сразу был виден.
    const inner = menu.querySelector(INNER_SELECTOR);
    if (inner && query) inner.scrollTop = 0;
  }

  function restoreItems(menu) {
    for (const item of getItems(menu)) {
      item.style.display = '';
    }
  }

  function firstVisibleItem(menu) {
    return getItems(menu).find(item => {
      if (getComputedStyle(item).display === 'none') return false;
      return visible(item);
    }) || null;
  }

  function clickItem(item) {
    if (!item) return;

    // Оставляем выбор самому ExtJS.
    for (const type of ['mouseover', 'mousedown', 'mouseup', 'click']) {
      item.dispatchEvent(
        new MouseEvent(type, {
          bubbles: true,
          cancelable: true,
          view: window,
          button: 0
        })
      );
    }
  }

  function enhance(menu) {
    if (!menu || wired.has(menu)) return;

    const inner = menu.querySelector(INNER_SELECTOR);
    if (!inner) return;

    const items = getItems(menu);
    if (items.length < 3) return;

    wired.add(menu);

    const search = document.createElement('div');
    search.className = SEARCH_CLASS;

    search.style.cssText = [
      'position:sticky',
      'top:0',
      'z-index:99999',
      'box-sizing:border-box',
      'width:100%',
      'padding:5px',
      'background:#fff',
      'border-bottom:1px solid #cfcfcf'
    ].join(';');

    const row = document.createElement('div');
    row.style.cssText = [
      'display:flex',
      'align-items:center',
      'gap:6px'
    ].join(';');

    const input = document.createElement('input');
    input.type = 'search';
    input.placeholder = 'Поиск…';
    input.autocomplete = 'off';

    input.style.cssText = [
      'box-sizing:border-box',
      'width:100%',
      'height:30px',
      'padding:4px 8px',
      'border:1px solid #aaa',
      'border-radius:4px',
      'outline:none',
      'background:#fff',
      'color:#222',
      'font:12px Arial,sans-serif'
    ].join(';');

    const count = document.createElement('span');
    count.className = 'uplink-extjs-search-count';
    count.style.cssText = [
      'flex:0 0 auto',
      'color:#777',
      'font:11px Arial,sans-serif',
      'white-space:nowrap'
    ].join(';');
    count.textContent = `${items.length} из ${items.length}`;

    row.append(input, count);
    search.append(row);

    // Вставляем внутрь именно scroll-контейнера ExtJS,
    // чтобы поиск был частью dropdown.
    inner.prepend(search);

    const stop = e => e.stopPropagation();

    for (const event of [
      'pointerdown',
      'mousedown',
      'mouseup',
      'click',
      'dblclick'
    ]) {
      input.addEventListener(event, stop);
      search.addEventListener(event, stop);
    }

    input.addEventListener('input', () => {
      applySearch(menu, input);
    });

    input.addEventListener('keydown', e => {
      e.stopPropagation();

      if (e.key === 'Escape') {
        e.preventDefault();

        if (input.value) {
          input.value = '';
          restoreItems(menu);
          count.textContent = `${getItems(menu).length} из ${getItems(menu).length}`;
        } else {
          input.blur();
        }

        return;
      }

      if (e.key === 'Enter') {
        e.preventDefault();
        clickItem(firstVisibleItem(menu));
        return;
      }

      if (e.key === 'ArrowDown') {
        e.preventDefault();

        const first = firstVisibleItem(menu);
        if (first) {
          first.scrollIntoView({ block: 'nearest' });
          first.dispatchEvent(
            new MouseEvent('mouseover', {
              bubbles: true,
              cancelable: true,
              view: window
            })
          );
        }
      }
    });

    // При каждом новом открытии dropdown старый запрос не должен оставаться.
    const menuObserver = new MutationObserver(() => {
      if (!visible(menu)) return;

      if (input.value) {
        input.value = '';
        restoreItems(menu);
        count.textContent = `${getItems(menu).length} из ${getItems(menu).length}`;
      }

      setTimeout(() => {
        if (visible(menu)) {
          input.focus({ preventScroll: true });
        }
      }, 0);
    });

    menuObserver.observe(menu, {
      attributes: true,
      attributeFilter: ['style', 'class']
    });

    // Если dropdown уже открыт в момент подключения.
    if (visible(menu)) {
      setTimeout(() => input.focus({ preventScroll: true }), 0);
    }
  }

  function scan() {
    for (const menu of document.querySelectorAll(MENU_SELECTOR)) {
      if (visible(menu)) enhance(menu);
    }
  }

  const observer = new MutationObserver(() => {
    clearTimeout(scanTimer);
    scanTimer = setTimeout(scan, 50);
  });

  observer.observe(document.documentElement, {
    childList: true,
    subtree: true,
    attributes: true,
    attributeFilter: ['style', 'class']
  });

  // ExtJS часто создаёт layer только после клика.
  document.addEventListener(
    'pointerdown',
    () => setTimeout(scan, 30),
    true
  );

  scan();
})();
